---- MODULE Nano ----
EXTENDS Naturals, FiniteSets

CONSTANTS
    Hash,          \* Set of all possible block hashes
    NoHashVal,     \* Sentinel hash value meaning "no previous hash"
    PrivateKey,    \* Set of private keys (not used directly)
    PublicKey,     \* Set of public keys (account identifiers)
    Node,          \* Set of network nodes
    GenesisBalance,\* Natural number: total coin supply
    NoBlockVal,    \* Sentinel block value meaning "no block"
    CalculateHash, \* External operator: CalculateHash(block, prevHash) -> Hash
    NoHash,        \* Another sentinel hash (for illustration)
    NoBlock        \* External constant representing the empty block

\* ----------------------------------------------------------------------
\* Block type definition
\* ----------------------------------------------------------------------
Block == [type      : {"Genesis","Send","Open","Receive","Change"},
          prev      : Hash,
          sender    : PublicKey,
          recipient : PublicKey,
          amount    : Nat,
          rep       : PublicKey,
          sig       : PublicKey]

\* ----------------------------------------------------------------------
\* Derived constants
\* ----------------------------------------------------------------------
BlockSet == [ ? IN {b \in Hash |-> some v} |-> v | v \in {Block, NoBlockVal}]
\* (used only in type invariant; not directly in the model)

\* ----------------------------------------------------------------------
\* Variables
\* ----------------------------------------------------------------------
VARIABLES
    LastHash,   \* Current global "last hash"
    Ledgers,    \* [n \in Node |-> [h \in Hash |-> {Block, NoBlockVal}]]
    Received    \* [n \in Node |-> SUBSET Hash]

\* ----------------------------------------------------------------------
\* Helper definitions
\* ----------------------------------------------------------------------
GetBlock(ledger, h) == IF h \in Domain(ledger) THEN ledger[h] ELSE NoBlockVal

\* Recursive balance computation for a single account on a given ledger
\* (the recursion stops when it reaches NoHashVal or a Genesis block)
Balance(ledger, acct, h) ==
    IF h = NoHashVal THEN 0
    ELSE
      LET blk == GetBlock(ledger, h) IN
        IF blk = NoBlockVal THEN 0
        ELSE
          IF blk.type = "Genesis" THEN GenesisBalance
          ELSE IF blk.type = "Send" THEN
                Balance(ledger, acct, blk.prev) - blk.amount
          ELSE IF blk.type = "Open" THEN
                (* redeem the referenced send block *)
                LET sendBlk == GetBlock(ledger, blk.prev) IN
                  Balance(ledger, acct, sendBlk.prev) + sendBlk.amount
          ELSE IF blk.type = "Receive" THEN
                Balance(ledger, acct, blk.prev) + blk.amount
          ELSE IF blk.type = "Change" THEN
                Balance(ledger, acct, blk.prev)
          ELSE 0

\* For a given ledger, the balance of an account at the current end of its chain
AccountBalance(ledger, acct) == Balance(ledger, acct, LastHash)

\* ----------------------------------------------------------------------
\* Type invariant (ensures all variables stay within their declared types)
\* ----------------------------------------------------------------------
TypeInvariant ==
    /\ LastHash \in Hash
    /\ Ledgers \in [n \in Node |-> [h \in Hash |-> {Block, NoBlockVal}]]
    /\ Received \in [n \in Node |-> SUBSET Hash]
    /\ \A n \in Node, h \in Hash :
          (h \in Domain(Ledgers[n]) /\ (Ledgers[n][h] \in Block \/ Ledgers[n][h] = NoBlockVal))
    /\ \A n \in Node : Received[n] \subseteq Hash

\* ----------------------------------------------------------------------
\* Cryptographic invariant (simplified: signature must equal the sender)
\* ----------------------------------------------------------------------
CryptographicInvariant ==
    \A n \in Node, h \in Domain(Ledgers[n]) :
        LET blk == Ledgers[n][h] IN
          IF blk \in Block THEN blk.sig = blk.sender ELSE TRUE

\* ----------------------------------------------------------------------
\* Balance invariant (send blocks cannot overdraw)
\* ----------------------------------------------------------------------
SendBalanceInvariant ==
    \A n \in Node, h \in Domain(Ledgers[n]) :
        LET blk == Ledgers[n][h] IN
          IF blk \in Block /\ blk.type = "Send" THEN
                blk.amount <= Balance(Ledgers[n], blk.sender, blk.prev)
          ELSE TRUE

\* Combine the safety properties
SafetyInvariant ==
    CryptographicInvariant /\ SendBalanceInvariant

\* ----------------------------------------------------------------------
\* Initial state
\* ----------------------------------------------------------------------
Init ==
    /\ LastHash = NoHashVal
    /\ Ledgers = [n \in Node |-> [h \in Hash |-> NoBlockVal]]
    /\ Received = [n \in Node |-> {}]

\* ----------------------------------------------------------------------
\* Actions
\* ----------------------------------------------------------------------
CreateGenesis ==
    /\ LastHash = NoHashVal
    /\ LET blk == [type      |-> "Genesis",
                   prev      |-> NoHashVal,
                   sender    |-> CHOOSE pk \in PublicKey : TRUE,
                   recipient |-> CHOOSE pk \in PublicKey : TRUE,
                   amount    |-> GenesisBalance,
                   rep       |-> CHOOSE pk \in PublicKey : TRUE,
                   sig       |-> CHOOSE pk \in PublicKey : TRUE]
        /\ h == CalculateHash(blk, NoHashVal)
    /\ LastHash' = h
    /\ Ledgers' = [n \in Node |-> [h' \in Hash |-> IF h' = h THEN blk ELSE NoBlockVal]]
    /\ UNCHANGED Received

CreateSend ==
    /\ LastHash \in Hash
    /\ LET n  == CHOOSE n \in Node : TRUE
        /\ pk == CHOOSE pk \in PublicKey : TRUE
        /\ r  == CHOOSE r \in PublicKey : TRUE
        /\ bal == Balance(Ledgers[n], pk, LastHash)
        /\ amt == CHOOSE amt \in Nat : amt <= bal /\ amt > 0
        /\ blk == [type      |-> "Send",
                   prev      |-> LastHash,
                   sender    |-> pk,
                   recipient |-> r,
                   amount    |-> amt,
                   rep       |-> pk,
                   sig       |-> pk]
        /\ h == CalculateHash(blk, LastHash)
    /\ LastHash' = h
    /\ Ledgers' = [n' \in Node |
          IF n' = n THEN [h' \in Hash |-> IF h' = h THEN blk ELSE NoBlockVal]
          ELSE Ledgers[n']]
    /\ Received' = [n' \in Node |-> Received[n'] \cup {h}]
    /\ UNCHANGED LastHash

CreateOpen ==
    /\ LastHash \in Hash
    /\ LET n  == CHOOSE n \in Node : TRUE
        /\ hSend == CHOOSE hSend \in Received[n] : TRUE
        /\ sendBlk == GetBlock(Ledgers[n], hSend)
        /\ pk == sendBlk.recipient
        /\ blk == [type      |-> "Open",
                   prev      |-> hSend,
                   sender    |-> pk,
                   recipient |-> pk,
                   amount    |-> 0,
                   rep       |-> pk,
                   sig       |-> pk]
        /\ h == CalculateHash(blk, hSend)
    /\ Ledgers' = [n' \in Node |
          IF n' = n THEN [h' \in Hash |-> IF h' = h THEN blk ELSE Ledgers[n'][h']]
          ELSE Ledgers[n']]
    /\ Received' = [n' \in Node |-> IF n' = n THEN Received[n'] \ {hSend} ELSE Received[n']]
    /\ UNCHANGED <<LastHash>>

CreateReceive ==
    /\ LastHash \in Hash
    /\ LET n  == CHOOSE n \in Node : TRUE
        /\ hSend == CHOOSE hSend \in Received[n] : TRUE
        /\ sendBlk == GetBlock(Ledgers[n], hSend)
