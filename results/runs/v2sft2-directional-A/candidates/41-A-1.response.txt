(* Type invariant *)
TypeOK ==
    /\ Susp \in [Proc -> SUBSET Proc]
    /\ TOut \in [Proc -> [Proc \ {p} -> Nat]]
    /\ LH \in [Proc -> [Proc \ {p} -> Nat]]
    /\ Clock \in [Proc -> Nat]
    /\ OutMsg \in [Proc -> SUBSET Messages]

(* Initial state *)
Init ==
    /\ Susp   = [p \in Proc |-> {}]
    /\ TOut   = [p \in Proc |-> [q \in Proc \ {p} |-> d0]]
    /\ LH     = [p \in Proc |-> [q \in Proc \ {p} |-> 0]]
    /\ Clock  = [p \in Proc |-> 0]
    /\ OutMsg = [p \in Proc |-> {}]

(* Derived constants *)

MaxThreshold(p) == MAX(SendPoint, PredictPoint, MaxTimeout(p))
