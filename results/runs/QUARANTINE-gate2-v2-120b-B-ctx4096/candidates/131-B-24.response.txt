  Cardinality(PositionsBefore(v, j))

  /\ seq \in Seq(Value)

vars == << seq, i, cand, cnt >>

\* ----------------------------------------------------------------------
Init ==
  /\ seq \in Seq(Value)
  /\ i = 1
  /\ cand = PickValue
  /\ cnt = 0

\* ----------------------------------------------------------------------

     /\ cnt' = cnt - 1
     /\ i' = i + 1
     /\ UNCHANGED << seq, cand >>

\* Specification
