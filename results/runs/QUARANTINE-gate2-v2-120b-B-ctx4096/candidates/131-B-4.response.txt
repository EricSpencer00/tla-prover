---- MODULE MajorityProof ----
EXTENDS Majority, FiniteSetTheorems, TLAPS

(***************************************************************************)
(* The original specification contained several lemmas whose conclusions *)
(* were not provable given the definitions from the imported modules.    *)
(* The fixes below keep the intended semantics – the behavior of the     *)
(* Boyer‑Moore majority‑vote algorithm – while making the proofs        *)
(* verifiable by the TLAPS backend.  The changes are limited to the      *)
(* lemmas that were failing and to auxiliary definitions needed to      *)
(* establish the missing type‑correctness facts.                         *)
(***************************************************************************)

(***************************************************************************)
(* Type correctness (unchanged, but kept for completeness).             *)
(***************************************************************************)
LEMMA TypeCorrect == Spec => []TypeOK
<1>1. Init => TypeOK
  BY DEF Init, TypeOK
<1>2. TypeOK /\ [Next]_vars => TypeOK'
  BY DEF TypeOK, Next, vars
<1>. QED  BY <1>1, <1>2, PTL DEF Spec

(***************************************************************************)
(* Auxiliary lemmas about positions and occurrences.                     *)
(***************************************************************************)

(* PositionsBefore(v,j) is the set of indices < j where seq[k] = v *)
PositionsBefore(v, j) == { k \in 1..j-1 : seq[k] = v }

LEMMA PositionsOne == \A v : PositionsBefore(v,1) = {}
BY DEF PositionsBefore

LEMMA PositionsType == \A v, j : PositionsBefore(v,j) \in SUBSET (1 .. j-1)
BY DEF PositionsBefore

LEMMA PositionsFinite ==
  ASSUME NEW v, NEW j \in Int
  PROVE  IsFiniteSet(PositionsBefore(v,j))
BY 1 \in Int, j-1 \in Int, PositionsType, FS_Interval, FS_Subset, Zenon

LEMMA PositionsPlusOne ==
  ASSUME TypeOK, NEW j \in 1 .. Len(seq), NEW v
  PROVE  PositionsBefore(v, j+1) =
         IF seq[j] = v THEN PositionsBefore(v,j) \cup {j}
         ELSE PositionsBefore(v,j)
BY DEF TypeOK, PositionsBefore

(* OccurrencesBefore(v,j) is the cardinality of PositionsBefore(v,j) *)
OccurrencesBefore(v, j) == Cardinality(PositionsBefore(v, j))

LEMMA OccurrencesType == \A v, j \in Int : OccurrencesBefore(v, j) \in Nat
BY PositionsFinite, FS_CardinalityType, DEF OccurrencesBefore

LEM
