---- MODULE Voting ----
EXTENDS Integers, TLAPS

CONSTANTS 
    Value,      \* Set of values that may be chosen.
    Acceptor,   \* Set of processes that can cast votes.
    Quorum      \* Set of quorums; each quorum is a subset of Acceptor.

(*-------------------------------------------------------------------*)
(*  Assumptions about quorums                                         *)
(*-------------------------------------------------------------------)
ASSUME QuorumAssumption == 
    /\ \A Q \in Quorum : Q \subseteq Acceptor
    /\ \A Q1, Q2 \in Quorum : Q1 \cap Q2 # {}

THEOREM QuorumNonEmpty == \A Q \in Quorum : Q # {}

(*-------------------------------------------------------------------*)
(*  Ballot numbers                                                    *)
(*-------------------------------------------------------------------)
Ballot == Nat

(*-------------------------------------------------------------------*)
(*  Variables                                                         *)
(*-------------------------------------------------------------------)
VARIABLES votes, maxBal

(*-------------------------------------------------------------------*)
(*  Type-correctness invariant                                        *)
(*-------------------------------------------------------------------)
TypeOK == 
    /\ votes   \in [Acceptor -> SUBSET (Ballot \X Value)]
    /\ maxBal  \in [Acceptor -> Ballot \cup {-1}]

(*-------------------------------------------------------------------*)
(*  Helper definitions                                               *)
(*-------------------------------------------------------------------)
VotedFor(a, b, v) == <<b, v>> \in votes[a]

ChosenAt(b, v) == \E Q \in Quorum : \A a \in Q : VotedFor(a, b, v)

chosen == { v \in Value : \E b \in Ballot : ChosenAt(b, v) }

DidNotVoteAt(a, b) == \A v \in Value : ~VotedFor(a, b, v)

CannotVoteAt(a, b) == 
    /\ maxBal[a] > b
    /\ DidNotVoteAt(a, b)

NoneOtherChoosableAt(b, v) ==
    \E Q \in Quorum :
        \A a \in Q : VotedFor(a, b, v) \/ CannotVoteAt(a, b)

SafeAt(b, v) == \A c \in 0..(b-1) : NoneOtherChoosableAt(c, v)

ShowsSafeAt(Q, b, v) ==
    /\ \A a \in Q : maxBal[a] >= b
    /\ \E c \in -1..(b-1) :
          /\ (c # -1) => \E a \in Q : VotedFor(a, c, v)
          /\ \A d \in (c+1)..(b-1), a \in Q : DidNotVoteAt(a, d)

(*-------------------------------------------------------------------)
