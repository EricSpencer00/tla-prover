---- MODULE MissionariesAndCannibals ----
(***************************************************************************)
(* This module specifies the classic missionaries and cannibals problem   *)
(* with three missionaries and three cannibals.  It has been corrected so *)
(* that TLC can find a solution without encountering a deadlock.           *)
(***************************************************************************)

EXTENDS Integers, FiniteSets

CONSTANTS Missionaries, Cannibals

VARIABLES bank_of_boat, who_is_on_bank

(***************************************************************************)
(* Type invariant.                                                         *)
(***************************************************************************)
TypeOK == /\ bank_of_boat \in {"E","W"}
          /\ who_is_on_bank \in [{"E","W"} -> SUBSET (Missionaries \cup Cannibals)]

(***************************************************************************)
(* Initial state: everything starts on the east bank.                     *)
(***************************************************************************)
Init == /\ bank_of_boat = "E"
        /\ who_is_on_bank = [i \in {"E","W"} |->
                               IF i = "E"
                               THEN Missionaries \cup Cannibals
                               ELSE {}]

(*
