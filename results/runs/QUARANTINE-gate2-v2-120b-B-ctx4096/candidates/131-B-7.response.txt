(***************************************************************************)
(* Auxiliary definitions for positions and occurrences.                  *)
(***************************************************************************)

\* PositionsBefore(v,j) is the set of indices < j where the element v occurs.
PositionsBefore(v, j) ==
  { k \in 1..j-1 : seq[k] = v }

\* The number of occurrences of v before position j.
OccurrencesBefore(v, j) ==
  Cardinality(PositionsBefore(v, j))

(***************************************************************************)
(* Type correctness lemma (unchanged).                                    *)
(***************************************************************************)

LEMMA TypeCorrect == Spec => []TypeOK
<1>1. Init => TypeOK
  BY DEF Init, TypeOK
<1>2. TypeOK /\ [Next]_vars => TypeOK'
  BY DEF TypeOK, Next, vars
<1>. QED  BY <1>1, <1>2, PTL DEF Spec

(***************************************************************************)
(* Lemmas about PositionsBefore.                                          *)
(***************************************************************************)

LEMMA PositionsOne == \A v : PositionsBefore(v, 1) = {}
BY DEF PositionsBefore

LEMMA PositionsType == \A v, j : PositionsBefore(v, j) \in SUBSET (1 .. j-1)
BY DEF PositionsBefore

LEMMA PositionsFinite ==
  ASSUME NEW v, NEW j \in Int
  PROVE  IsFiniteSet(PositionsBefore(v, j))
PROOF
  CASE j <= 1
    THEN
      PositionsBefore(v, j) = {}
      BY DEF PositionsBefore, Nat, SubsetSeq, FiniteSets
      QED
  CASE j > 1
    THEN
      PositionsBefore(v, j) \subseteq 1..j-1
        BY DEF PositionsBefore, PositionsType
      QED
  QED
BY 1 \in Nat, j-1 \in Nat, PositionsType, FS_Interval, FS_Subset, Zenon

LEMMA PositionsPlusOne ==
  ASSUME TypeOK,
         NEW j \in 1 .. Len(seq),
         NEW v
  PROVE  PositionsBefore(v, j+1) =
         IF seq[j] = v THEN PositionsBefore(v, j) \cup {j}
         ELSE PositionsBefore(v, j)
PROOF
  (* Direct expansion of the definition *)
  BY DEF PositionsBefore
QED
