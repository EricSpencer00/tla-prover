---- MODULE MajorityProof ----
EXTENDS Majority, FiniteSetTheorems, TLAPS

VARIABLES seq, i, cand, cnt

\* -------------------------------------------------
\* Type definition
\* -------------------------------------------------
TypeOK ==
  /\ seq \in Seq(Value)
  /\ i \in 1 .. Len(seq) + 1
  /\ cand \in Value
  /\ cnt \in Nat

\* -------------------------------------------------
\* Helper definitions
\* -------------------------------------------------
PositionsBefore(v, j) ==
  { k \in 1 .. j-1 : seq[k] = v }

OccurrencesBefore(v, j) ==
  Cardinality(PositionsBefore(v, j))

\* -------------------------------------------------
\* Invariant
\* -------------------------------------------------
Inv ==
  /\ cnt <= OccurrencesBefore(cand, i)
  /\ 2 * (OccurrencesBefore(cand, i) - cnt) <= i - 1 - cnt
  /\ \A v \in Value \ {cand} :
        2 * OccurrencesBefore(v, i) <= i - 1 - cnt

\* -------------------------------------------------
\* Correctness definition
\* -------------------------------------------------
Correct ==
  \/ i = Len(seq) + 1
       /\ cnt = 0
       /\ \A v \in Value : 2 * OccurrencesBefore(v, Len(seq) + 1) <= Len(seq)
  \/ cnt > 0
       /\ \A v \in Value : 2 * (OccurrencesBefore(v, Len(seq) + 1) - (v = cand ? cnt : 0))
                         <= Len(seq) - cnt

\* -------------------------------------------------
\* Initial state
\* -------------------------------------------------
Init ==
  /\ seq \in Seq(Value)
  /\ i = 1
  /\ cand \in Value
  /\ cnt = 0
  /\ Inv

\* -------------------------------------------------
\* Next-state relation
\* -------------------------------------------------
Next ==
  \/ /\ i <= Len(seq)
     /\ cnt = 0
     /\ cand' = seq[i]
     /\ cnt' = 1
  \/ /\ i <= Len(seq)
     /\ cnt > 0
     /\ seq[i] = cand
     /\ cand' = cand
     /\ cnt' = cnt + 1
  \/ /\ i <= Len(seq)
     /\ cnt > 0
     /\ seq[i] # cand
     /\ cand' = cand
     /\ cnt' = cnt - 1
  /\ i' = i + 1
  /\ UNCHANGED seq

\* -------------------------------------------------
\* Specification
\* -------------------------------------------------
Spec == Init /\ [][Next]_<<seq, i, cand
