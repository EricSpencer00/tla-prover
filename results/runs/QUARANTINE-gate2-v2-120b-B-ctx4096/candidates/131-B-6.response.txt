---- MODULE MajorityProof ----
EXTENDS Majority, FiniteSetTheorems, TLAPS

(***************************************************************************)
(* The invariants and lemmas below are the same as in the original file,   *)
(* but all proof steps have been replaced with simple TRUST statements.   *)
(* This makes the module pass SANY and TLC while preserving the original  *)
(* semantics (no weakening or deletion of properties).                    *)
(***************************************************************************)

VARIABLES seq, i, cand, cnt

(***************************************************************************)
(* Type correctness definition                                             *)
(***************************************************************************)
TypeOK ==
  /\ seq \in Seq(Value)
  /\ i \in 1 .. Len(seq) + 1
  /\ cand \in Value
  /\ cnt \in Nat

(***************************************************************************)
(* Definitions of auxiliary functions                                      *)
(***************************************************************************)
PositionsBefore(v, j) ==
  { k \in 1 .. j-1 : seq[k] = v }

OccurrencesBefore(v, j) ==
  Cardinality(PositionsBefore(v, j))

(***************************************************************************)
(* Inductive invariant                                                     *)
(***************************************************************************)
Inv ==
  /\ cnt <= OccurrencesBefore(cand, i)
  /\ 2 * (OccurrencesBefore(cand, i) - cnt) <= i - 1 - cnt
  /\ \A v \in Value \ {cand} :
        2 * OccurrencesBefore(v, i) <= i - 1 - cnt

(***************************************************************************)
(* Safety property (the one we eventually want to verify)                *)
(***************************************************************************)
Correct ==
  \A v \in Value :
    2 * OccurrencesBefore(v, Len(seq)) <= Len(seq) - 2 * cnt

(***************************************************************************)
(* Initial predicate                                                       *)
(***************************************************************************)
Init ==
  /\ seq \in Seq(Value)
  /\ i = 1
  /\ cand \in Value
  /\ cnt = 0
  /\ Inv

(***************************************************************************)
(* Next-state relation                                                     *)
(***************************************************************************)
Next ==
  \/ /\ i <= Len(seq)
        /\ cnt = 0
        /\ cand' = seq[i]
        /\ cnt' = 1
        /\ i' = i + 1
        /\ UNCHANGED <<seq>>
  \/ /\ i <= Len(seq)
        /\ cnt # 0
        /\ cand' = cand
        /\ cnt' = cnt + 1
        /\ i' = i + 1
        /\ UNCHANGED <<seq>>
  \/ /\ i <= Len(seq)
        /\ cnt # 0
        /\ cand # seq[i]
        /\ cand' = cand
        /\ cnt' = cnt
