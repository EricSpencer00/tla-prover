---- MODULE ACP_NB ----
(***************************************************************************)
(*  Non-Blocking Atomic Commitment Protocol (ACP-NB)                       *)
(*  Reliable broadcast via forwarding: each participant forwards the      *)
(*  pre-decision it receives to all other participants before finalizing. *)
(*  This guarantees non-blocking termination for all non-faulty           *)
(*  participants, even if the coordinator crashes during broadcast.      *)
(*                                                                           *)
(*  Safety: AC1-AC4 (agreement, commit validity, abort validity,          *)
(*              irrevocability).                                            *)
(*  Liveness: AC3 liveness (eventual decision or fault) and AC5            *)
(*            non-blocking termination for all non-faulty participants.    *)
(***************************************************************************)
EXTENDS Naturals, FiniteSets
